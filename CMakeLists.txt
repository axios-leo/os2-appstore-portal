# Generated packaging build; upstream module sources remain byte-for-byte unchanged.
cmake_minimum_required(VERSION 3.16)
project(os2_appstore_portal_handoff LANGUAGES CXX)
if(NOT UNIX)
  message(FATAL_ERROR "This source baseline requires Linux/POSIX. Use Linux or WSL Ubuntu.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Werror)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" OS2_RELEASE)
string(STRIP "${OS2_RELEASE}" OS2_RELEASE)
add_compile_definitions(OS2_RELEASE="${OS2_RELEASE}" OS2_BUILD_ID="4a650bc5")
enable_testing()
add_library(os2_platform INTERFACE)
target_include_directories(os2_platform INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/platform/include")
add_executable(test_platform platform/tests/test_platform.cpp)
target_link_libraries(test_platform PRIVATE os2_platform)
add_test(NAME test_platform COMMAND test_platform)
add_subdirectory(modules/appstore-portal)
